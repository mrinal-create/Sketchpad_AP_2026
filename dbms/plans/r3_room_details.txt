-- Q3 (R3): Room Details & Collaborator Profile List
-- EXPLAIN ANALYZE command:

EXPLAIN ANALYZE
SELECT 
  r."title",
  r."ownerId",
  COALESCE(
    (
      SELECT json_agg(
        jsonb_build_object(
          'user', jsonb_build_object(
            'id', u."id",
            'name', u."name",
            'email', u."email",
            'emailVerified', u."emailVerified",
            'image', u."image",
            'createdAt', u."createdAt",
            'updatedAt', u."updatedAt"
          )
        )
      )
      FROM "RoomInvite" ri
      INNER JOIN "user" u ON u."id" = ri."userId"
      WHERE ri."roomId" = r."id"
    ),
    '[]'::json
  ) AS "roomInvites"
FROM "Room" r
WHERE r."id" = 'clx_room_9876543210ab';
