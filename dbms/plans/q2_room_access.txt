-- Q2 (R2): Room Access Authorization Check (Liveblocks Auth)
-- EXPLAIN ANALYZE command:

EXPLAIN ANALYZE
SELECT r."id"
FROM "Room" r
WHERE r."id" = 'clx_room_9876543210ab'
  AND (
    r."ownerId" = 'usr_clx9a01bc000108l41b2c3d4e'
    OR EXISTS (
      SELECT 1
      FROM "RoomInvite" ri
      WHERE ri."roomId" = r."id"
        AND ri."userId" = 'usr_clx9a01bc000108l41b2c3d4e'
    )
  )
LIMIT 1;
