-- Q1 (R1): Dashboard Load - User Profile, Owned Rooms & Room Invites
-- EXPLAIN ANALYZE command:

EXPLAIN ANALYZE
SELECT 
  u."id",
  u."name",
  u."email",
  u."emailVerified",
  u."image",
  u."createdAt",
  u."updatedAt",
  COALESCE(
    (
      SELECT json_agg(
        jsonb_build_object(
          'id', r."id",
          'ownerId', r."ownerId",
          'createdAt', r."createdAt",
          'title', r."title"
        )
      )
      FROM "Room" r
      WHERE r."ownerId" = u."id"
    ),
    '[]'::json
  ) AS "ownedRooms",
  COALESCE(
    (
      SELECT json_agg(
        jsonb_build_object(
          'id', ri."id",
          'roomId', ri."roomId",
          'userId', ri."userId",
          'room', jsonb_build_object(
            'id', ir."id",
            'ownerId', ir."ownerId",
            'createdAt', ir."createdAt",
            'title', ir."title"
          )
        )
      )
      FROM "RoomInvite" ri
      INNER JOIN "Room" ir ON ir."id" = ri."roomId"
      WHERE ri."userId" = u."id"
    ),
    '[]'::json
  ) AS "roomInvites"
FROM "user" u
WHERE u."id" = 'usr_clx9a01bc000108l41b2c3d4e';
