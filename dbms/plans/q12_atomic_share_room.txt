-- Q12 (W5): Atomic Share Room - Verify Ownership & Insert Invite via CTE
-- EXPLAIN ANALYZE command:

EXPLAIN ANALYZE
WITH authorized_room AS (
  SELECT "id"
  FROM "Room"
  WHERE "id" = 'clx_room_9876543210ab'
    AND "ownerId" = 'usr_clx9a01bc000108l41b2c3d4e'
),
target_user AS (
  SELECT "id"
  FROM "user"
  WHERE "email" = 'collaborator@sketchpad.dev'
)
INSERT INTO "RoomInvite" ("id", "roomId", "userId")
SELECT 'inv_clx_9988776655aa', authorized_room."id", target_user."id"
FROM authorized_room, target_user
RETURNING "id", "roomId", "userId";

-- Single atomic CTE operation that validates room ownership, looks up the invitee by email, and inserts
